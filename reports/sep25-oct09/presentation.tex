\section{CMA-ES Optimisation}

\begin{frame}{CMA-ES: the algorithm}
  \begin{columns}[c]
    \begin{column}{1.0\textwidth}
      Keeps a \textbf{search cloud} $\mathcal{N}(m, \sigma^2 C)$ of parameter sets:
      \begin{itemize}\small
        \item centre $m$ \hfill 
        \item size $\sigma$ \hfill 
        \item shape $C$ \hfill 
      \end{itemize}
      \vspace{0.4em}
      \textbf{Every generation}
      \begin{enumerate}\small
        \item \textbf{Sample}: sample 15 parameter sets from the cloud
        \item \textbf{Select}: Score each sets and keep the best 7
        \item \textbf{Move}: centre $\leftarrow$ weighted average of the 7
        \item \textbf{Update}: $\sigma$ and $C$ 
      \end{enumerate}
    \end{column}

  \end{columns}
\end{frame}

\begin{frame}{Differential Evolution (DE)}
  \begin{columns}[c]
    \begin{column}{1.0\textwidth}
      Keeps a {population} of parameter sets $x_1, \dots, x_{N}$ spread over the search box
      \vspace{0.4em}

      \textbf{Every generation, for each member $x_i$}
      \begin{enumerate}\small
        \item \textbf{Mutate}: start at the best member, add scaled differences of random members
              \[ v = x_{i} + F\,\big[(x_{r1} - x_{r2}) + (x_{r3} - x_{r4})\big] \]
        \item \textbf{Crossover}: trial $u$ takes a block of coordinates from $v$, the rest from $x_i$
        \item \textbf{Select}: score $u$ ; $u$ replaces $x_i$ {only if it is better}
      \end{enumerate}
      \vspace{0.3em}

    \end{column}

  \end{columns}
\end{frame}

\begin{frame}{Capturing of state}

        \item TTC of vehicle front left/right
        \item TTC of vehicles rear (all), sides

\begin{frame}{Feature definitions}
  \centering
  \small
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{ll}
    \toprule
    \textbf{Feature} & \textbf{Definition} \\
    \midrule
    i)\ \ TTC ahead        & $1 - e^{-1/\mathrm{TTC}}$ \\
    ii)\ TTC front side    & $1 - e^{-1/\mathrm{TTC}}$ \\

    v)\ \ Gap front side   & $e^{-g_x/20\,\text{m}}$ \\
    vi)\ Road clearance    & $e^{-r/1\,\text{m}}$ \\
    vii) Speed             & $v_x / 34$ \\
    viii) Density          & $1 - e^{-n/10}$ \\
    \bottomrule
  \end{tabular}

\end{frame}

\begin{frame}{Adaptive policy}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \textbf{Policy:} $W \in \mathbb{R}^{4 \times 8}$, $b \in \mathbb{R}^{4}$ \quad (36 params)
      \begin{align*}
        z_t^* &= \mathrm{clip}\big(W f_t + b,\ -\ln 10,\ +\ln 10\big) \\[2pt]
        z_t   &= z_{t-1} + \frac{\Delta t}{T}\,\big(z_t^* - z_{t-1}\big) \\[2pt]
        p_t   &= p_0\, e^{z_t}
      \end{align*}
      \[
        p = \big(\lambda_{obs},\ \alpha,\ \tau_y,\ \delta_{lane}\big)
      \]
    \end{column}
    
  \end{columns}
\end{frame}

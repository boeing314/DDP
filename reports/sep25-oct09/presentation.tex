\subsection{$\tau$ switch}

\begin{frame}{Change I : sensitivity to $\tau_x, \tau_y$}
  \centering
  \includegraphics[width=0.8\textwidth]{../sep18-25/tau_sensitivity.png}
  \begin{itemize}\small
    \item The shape of the potential is highly sensitive to $\tau_y/\tau_x$.
    \item If $\tau_y \ll \tau_x$ the $\Delta y^2/\tau_y^2$ term dominates $\Rightarrow$
          \textbf{no lateral force} ahead and behind; same on the sides when $\tau_y/\tau_x \gg 1$.
    \item \textbf{Fix:} switch $\tau$ with region of $k$
  \end{itemize}
\end{frame}

\subsection{Modified $\tau$}

\begin{frame}{Change II: reformulating $\tau$}
  \begin{columns}[c]
    \begin{column}{0.45\textwidth}
      \centering
      \includegraphics[width=0.9\linewidth]{../sep18-25/vel_original.png}\par
      {\footnotesize Force vs $\Delta v_x$: kink from the $\tau$ term}
    \end{column}
    \begin{column}{0.53\textwidth}
      Current form:
      {\small
      \[
        \begin{aligned}
        \tau = \Big[&(1+\alpha|\Delta v_x|) \\
               &+ (\alpha|\Delta v_x|-1)\tanh(-\Delta x \Delta v_x)\Big]\,\tau_x
        \end{aligned}
      \]}
      \begin{itemize}\small
        \item Approaching gives $\tau = \alpha\tau_x|\Delta v_x|$, which can be \emph{smaller} than $\tau_x$
        \item \textbf{Fix:} Minimum value of $\tau$ when vehicles approaches should be $\tau_x$:
      \end{itemize}
      \[
        \tau =
        \begin{cases}
          \tau_x & \text{receding} \\
          \tau_x(1+\alpha|\Delta v_x|) & \text{approaching}
        \end{cases}
      \]
    \end{column}
  \end{columns}
\end{frame}

\section{SUMO Implementation}

\begin{frame}{Step 1: Change of frame}
  Rotate the request onto the car's heading, then scale it onto the grip ellipse.
  \begin{align*}
    a_{lon} &= \cos\psi\, a_x + \sin\psi\, a_y \\
    a_{lat} &= -\sin\psi\, a_x + \cos\psi\, a_y
  \end{align*}
  \[
    \left(\frac{a_{lon}}{A_{lon}}\right)^2 + \left(\frac{a_{lat}}{A_{lat}}\right)^2 \le 1
  \]
  \[
    A_{lon} = \begin{cases} a_{acc} & a_{lon} \ge 0 \\ a_{brake} & a_{lon} < 0 \end{cases}
  \]
  \begin{itemize}\small
    \item Scale as needed if term is greater than 1.
  \end{itemize}
\end{frame}

\begin{frame}{Steps 2,3: speed limit and jerk limits}
  \begin{columns}[c]
    \begin{column}{0.75\textwidth}
      \begin{align*}
        a_{lon} &\le \frac{v_{limit} - v}{\tau_v} \\[6pt]
        \big|a_{lon}^{k+1} - a_{lon}^{k}\big| &\le j\,\Delta t
      \end{align*}
    \end{column}
   
  \end{columns}
\end{frame}

\begin{frame}{Step 4: Curvature limits}
  \begin{align*}
    \kappa &= \frac{a_{lat}}{v^2} \\
    |\kappa| &\le \min\!\left(\kappa_{max},\ \frac{A_{lat}}{v^2}\right) \\
    |\Delta\kappa| &\le \min\!\left(\dot{\kappa}_{max},\ \frac{j_{lat}}{v^2}\right)\Delta t
  \end{align*}
\end{frame}

\section{Adaptive Model}

\begin{frame}{Adaptive framework}
  \vfill
  \centering
  \includegraphics[width=0.85\textwidth]{adaptive_framework.png}
  \vspace{1em}

  \begin{itemize}\small
    \item A single fixed DIPF parameter set is a compromise over all traffic situations
    \item \textbf{Input}: a compact description of the traffic around the ego, every $0.1$\,s
    \item \textbf{Adaptive model}: decides how the DIPF parameters should change for that situation
    \item \textbf{Output}: the DIPF parameters used by the controller at that step
  \end{itemize}
  \vfill
\end{frame}

\begin{frame}{State}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\linewidth]{Surrounding_division.png}

    \begin{column}{0.5\textwidth}
      \textbf{8 features}
      \begin{enumerate}[i)]
        \item TTC of vehicle ahead
        \item TTC of vehicle front left/right
        \item TTC of vehicles rear (all), sides

\begin{frame}{Adaptive policy: the idea}
  \begin{itemize}
    \item We already have a good base set $p_0=\big(\lambda_{obs,0},\ \alpha_0,\ \tau_{y,0},\ \delta_{lane,0}\big)$
    \item Rather than predicting the parameters from scratch we rescale the current values:
          \[ p_t = p_0 \cdot m_t, \qquad m_t > 0 \]
    \item The multiplier $m_t$ depends on the state $f_t$ and is learned.
    \item We write the multiplier as an exponential, $m_t = e^{z_t}$
    \end{itemize}
\end{frame}

\begin{frame}{Adaptive policy: mapping state to multiplier}
  We need a model $g$ that maps the state to one exponent per parameter:
  \[
    g:\ f_t \in [0,1]^8 \;\longrightarrow\; z_t \in \mathbb{R}^4
  \]
  \textbf{A simple linear model:}
  \[
    z_t = W f_t + b, \qquad W \in \mathbb{R}^{4 \times 8},\ b \in \mathbb{R}^{4} \quad (36 \text{ params})
  \]
  \[
    p_{k,t} = p_{k,t}\, e^{\,b_k + \sum_j w_{kj} f_j}
  \]

\end{frame}

\begin{frame}{Adaptive policy: in depth}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      Every control step ($\Delta t = 0.1$\,s):
      \begin{align*}
        z_t^* &= \mathrm{clip}\big(W f_t + b,\ -\ln 10,\ +\ln 10\big) \\[4pt]
        z_t   &= z_{t-1} + \frac{\Delta t}{T}\,\big(z_t^* - z_{t-1}\big) \\[4pt]
        p_t   &= p_0\, e^{z_t}
      \end{align*}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{enumerate}\small
        \item \textbf{Bounded params}: Output is clipped to $[0.1\,p_0,\ 10\,p_0]$
        \item \textbf{Smoothing}: avoids big changes in the parameters
      \end{enumerate}
      \vspace{0.3em}
    \end{column}
  \end{columns}
\end{frame}
